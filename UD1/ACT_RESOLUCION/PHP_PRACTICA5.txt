Ejercicio 4
Crea un array multidimensional $productos donde cada elemento sea un array asociativo con nombre, precio y categoria (al menos 4 productos, 2 categorías distintas). Recorre el array con foreach y muestra solo los productos de una categoría concreta, junto con el precio total de esos productos filtrados.









<?php
// Primer producto (reutilizado del ejercicio anterior)
$producto1 = [
    "nombre" => "Teclado Mecánico",
    "precio" => 49.99,
    "stock"  => 5
];

// Segundo producto
$producto2 = [
    "nombre" => "Ratón Inalámbrico",
    "precio" => 25.50,
    "stock"  => 3
];

// Cálculo del coste total si se agota todo el stock
$subtotalProd1 = $producto1["precio"] * $producto1["stock"];
$subtotalProd2 = $producto2["precio"] * $producto2["stock"];
$totalGeneral  = $subtotalProd1 + $subtotalProd2;

echo "<h3>Cálculo de compra de stock total:</h3>";
echo $producto1["nombre"] . ": " . $producto1["stock"] . " x " . $producto1["precio"] . " € = " . number_format($subtotalProd1, 2) . " €<br>";
echo $producto2["nombre"] . ": " . $producto2["stock"] . " x " . $producto2["precio"] . " € = " . number_format($subtotalProd2, 2) . " €<br>";
echo "<strong>Precio total combinado: " . number_format($totalGeneral, 2) . " €</strong><br>";
?>



<?php
// Array multidimensional: array que contiene arrays asociativos
$productos = [
    [
        "nombre"    => "Monitor 24 pulgadas",
        "precio"    => 129.90,
        "categoria" => "Electrónica"
    ],
    [
        "nombre"    => "Camiseta deportiva",
        "precio"    => 19.95,
        "categoria" => "Ropa"
    ],
    [
        "nombre"    => "Auriculares Bluetooth",
        "precio"    => 35.00,
        "categoria" => "Electrónica"
    ],
    [
        "nombre"    => "Pantalón vaquero",
        "precio"    => 39.90,
        "categoria" => "Ropa"
    ]
];

// Categoría que vamos a filtrar
$categoriaFiltrar = "Electrónica";
$totalCategoria = 0;

echo "<h3>Productos de la categoría: " . $categoriaFiltrar . "</h3>";
echo "<ul>";

foreach ($productos as $item) {
    if ($item["categoria"] === $categoriaFiltrar) {
        echo "<li>" . $item["nombre"] . " - " . number_format($item["precio"], 2) . " €</li>";
        // Acumulación del precio para el total
        $totalCategoria += $item["precio"];
    }
}

echo "</ul>";
echo "<strong>Precio total de la categoría " . $categoriaFiltrar . ":</strong> " . number_format($totalCategoria, 2) . " €";
?>
